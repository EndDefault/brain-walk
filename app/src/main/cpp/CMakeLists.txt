cmake_minimum_required(VERSION 3.22.1)
# The Windows build host crashes re-searching cached NDK programs. Preserve
# explicitly configured tools; delegate all actual searches to CMake.
if(CMAKE_HOST_WIN32)
    macro(find_program variable)
        if(NOT DEFINED ${variable} OR "${${variable}}" MATCHES "-NOTFOUND$")
            _find_program(${variable} ${ARGN})
        endif()
    endmacro()
endif()
project(memory_author LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(llama
    GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
    GIT_TAG 5e03bdd8700948b9c41c54dd1b00f28a2aebc03f)
get_filename_component(LOCAL_LLAMA "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.local-tools/llama.cpp" ABSOLUTE)
if(EXISTS "${LOCAL_LLAMA}/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_LLAMA "${LOCAL_LLAMA}")
endif()
FetchContent_MakeAvailable(llama)
add_library(memory_author SHARED question_author.cpp)
target_link_libraries(memory_author PRIVATE llama android log)
target_compile_options(memory_author PRIVATE -O3)
target_link_options(memory_author PRIVATE "-Wl,-z,max-page-size=16384")
